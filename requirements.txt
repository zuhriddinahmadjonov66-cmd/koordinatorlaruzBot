def load_env():
    dotenv_path = ROOT / ".env"
    if dotenv_path.exists():
        for line in dotenv_path.read_text(encoding="utf-8-sig").splitlines():
            if not line or line.strip().startswith("#") or "=" not in line:
                continue
            key, value = line.split("=", 1)
            os.environ.setdefault(key.strip(), value.strip().strip('"').strip("'"))


def safe_int(value, default=0):
    try:
        return int(value)
    except (TypeError, ValueError):
        return default


class Telegram:
    def __init__(self, token):
        self.base = "https://api.telegram.org/bot" + token + "/"

    def call(self, method, **data):
        body = json.dumps(data).encode("utf-8")
        req = urllib.request.Request(
            self.base + method,
            data=body,
            headers={"Content-Type": "application/json"},
        )
        try:
            with urllib.request.urlopen(req, timeout=45) as response:
                result = json.load(response)
        except urllib.error.HTTPError as exc:
            raise RuntimeError(f"Telegram HTTP {exc.code}") from None
        except (urllib.error.URLError, TimeoutError):
            raise RuntimeError("Telegram bilan aloqa yo‘q") from None
        if not result.get("ok"):
            raise RuntimeError("Telegram so‘rovni qabul qilmadi")
        return result["result"]

    def ensure_admin(self):
        existing = self.one("SELECT id FROM users WHERE id=?", (self.owner,))
        if not existing:
            self.write(
                "INSERT INTO users(id, name, role, approved, region, district, school, direction) VALUES(?,?,?,?,?,?,?,?)",
                (self.owner, "Bosh administrator", "owner", 1, "", "", "", ""),
            )
        else:
            self.write(
                "UPDATE users SET name=COALESCE(name,'Bosh administrator'), role='owner', approved=1 WHERE id=?",
                (self.owner,),
            )

    def one(self, sql, args=()):
        return self.db.execute(sql, args).fetchone()

    def rows(self, sql, args=()):
        return self.db.execute(sql, args).fetchall()

    def write(self, sql, args=()):
        cur = self.db.execute(sql, args)
        self.db.commit()
        return cur.lastrowid

    def send(self, uid, text, keyboard=None, disable_web_page_preview=True):
        if keyboard is None:
            keyboard = MAIN_MENU
        try:
            for start in range(0, len(text), 3800):
                self.api.call(
                    "sendMessage",
                    chat_id=uid,
                    text=text[start : start + 3800],
                    reply_markup={"keyboard": keyboard, "resize_keyboard": True},
                    disable_web_page_preview=disable_web_page_preview,
                )
        except RuntimeError:
            pass

    def notify(self, uid, text):
        if uid is None:
            return
        try:
            self.send(uid, text)
        except Exception:
            pass

    def user(self, uid):
        return self.one("SELECT * FROM users WHERE id=?", (uid,))

    def session(self, uid, kind, data):
        self.write("INSERT OR REPLACE INTO sessions(id, kind, data) VALUES(?,?,?)", (uid, kind, json.dumps(data, ensure_ascii=False)))

    def clear_session(self, uid):
        self.write("DELETE FROM sessions WHERE id=?", (uid,))

    def get_session(self, uid):
        row = self.one("SELECT * FROM sessions WHERE id=?", (uid,))
        if not row:
            return None
        data = row["data"]
        try:
            return row["kind"], json.loads(data or "{}")
        except Exception:
            return row["kind"], {}

    def is_manager(self, user_row):
        if not user_row or not user_row["approved"]:
            return False
        return user_row["role"] in ("owner", "region", "district")

    def is_owner(self, user_row):
        return user_row and user_row["role"] == "owner" and user_row["approved"]

    def has_pro(self, uid):
        u = self.user(uid)
        if not u:
            return False
        if not u["pro_active"]:
            return False
        if not u["pro_expires_at"]:
            return False
        try:
            expires = datetime.strptime(u["pro_expires_at"], "%Y-%m-%d %H:%M:%S").replace(tzinfo=TZ)
            return datetime.now(TZ) < expires
        except Exception:
            return False

    def is_visible_task(self, user_row, task_row):
        if not user_row or not user_row["approved"]:
            return False
        if not task_row:
            return False
        if user_row["role"] == "owner":
            return True
        if task_row["region"] and task_row["region"] != user_row["region"]:
            return False
        if task_row["district"] and task_row["district"] != user_row["district"]:
            return False
        if task_row["direction"] and task_row["direction"] != user_row["direction"]:
            return False
        return True

    def can_review_task(self, user_row, task_row):
        if not self.is_manager(user_row):
            return False
        if user_row["role"] == "owner":
            return True
        if task_row["region"] != user_row["region"]:
            return False
        if user_row["role"] == "region":
            return True
        if user_row["role"] == "district":
            return task_row["district"] == user_row["district"]
        return False

    def get_rating_summary(self):
        rows = self.rows(
            """
            SELECT AVG(score) AS avg_score, direction FROM reports
            WHERE score IS NOT NULL AND score BETWEEN 1 AND 10
            GROUP BY direction
            ORDER BY avg_score DESC
            """
        )
        overall = self.one("SELECT AVG(score) AS avg_score FROM reports WHERE score IS NOT NULL AND score BETWEEN 1 AND 10")
        return overall, rows

    def handle(self, message):
        if message.get("chat", {}).get("type") != "private":
            return
        uid = int(message.get("from", {}).get("id"))
        text = (message.get("text") or "").strip()
        u = self.user(uid)
        command = text.split()[0].split("@") [0] if text else ""

        if command == "/id":
            self.send(uid, f"Telegram ID: {uid}")
            return

        if text in ("❌ Bekor qilish", "/cancel"):
            self.clear_session(uid)
            self.send(uid, "Bekor qilindi.")
            return

        if command == "/start":
            self.clear_session(uid)
            if not u:
                self.session(uid, "register_name", {})
                self.send(uid, "Assalomu alaykum! Botga ro‘yxatdan o‘tish uchun ismingizni kiriting:")
                return
            if not u["approved"]:
                self.send(uid, "Arizangiz tasdiqlanishini kutyapti. Iltimos, admin roziligini kuting.")
                return
            self.send(uid, "Bosh menyu", self.keyboard_for_main(u))
            return

        if command == "/help":
            self.send(
                uid,
                "\n".join(
                    [
                        "Botdan foydalanish bo‘yicha ko‘rsatma:",
                        "/start — bosh menyu",
                        "/cancel — bekor qilish",
                        "/id — Telegram ID ni ko‘rish",
                        "/report 12 — topshiriq bo‘yicha hisobot yuborish",
                        "/promo 123 — promo kodni kiritish",
                        "/delete_data — ma’lumotlarni o‘chirish so‘rovi",
                        "/help — yordam",
                    ]
                ),
            )
            return

        if command == "/delete_data":
            self.write(
                "INSERT OR IGNORE INTO data_delete_requests(user_id) VALUES(?)",
                (uid,),
            )
            self.notify(self.owner, f"Ma’lumotlarni o‘chirish so‘rovi: {uid}")
            self.send(uid, "Ma’lumotlar o‘chirish so‘rovi adminga yuborildi. Admin tasdiqlagach, ma’lumotlaringiz o‘chiriladi.")
            return

        if command == "/promo":
            code = text.split(maxsplit=1)[1].strip() if len(text.split()) > 1 else ""
            self.process_promo(uid, code)
            return

        session = self.get_session(uid)
        if not u:
            self.send(uid, "Avval /start bosing.")
            return

        if not u["approved"]:
            self.send(uid, "Arizangiz hali tasdiqlanmagan. Adminning tasdig‘i kutilmoqda.")
            return

        if session:
            kind, payload = session
            if kind == "register_name":
                name = text.strip()
                if not name or len(name) > 80:
                    self.send(uid, "Iltimos, 1–80 belgidan iborat ismingizni kiriting.")
                    return
                self.write("UPDATE users SET name=? WHERE id=?", (name, uid))
                self.session(uid, "register_region", {"name": name})

            if kind == "register_region":
                region = text.strip()
                if region not in REGIONS:
                    self.send(uid, "Viloyatni ro‘yxatdan tanlang.", [[r] for r in REGIONS])
                    return
                self.write("UPDATE users SET region=? WHERE id=?", (region, uid))
                self.session(uid, "register_district", {"region": region})
                self.send(uid, "Tuman/shahar nomini yozing:")
                return
            if kind == "register_district":
                district = text.strip()
                if not district or len(district) > 80:
                    self.send(uid, "Tuman yoki shahar nomini to‘g‘ri kiriting.")
                    return
                self.write("UPDATE users SET district=? WHERE id=?", (district, uid))
                self.session(uid, "register_school", {})
                self.send(uid, "Maktab yoki tashkilot nomini yozing:")

            if kind == "register_school":
                school = text.strip()
                if not school or len(school) > 150:

                    return
                self.write("UPDATE users SET school=? WHERE id=?", (school, uid))
                self.session(uid, "register_direction", {})
                self.send(uid, "Yo‘nalishingizni tanlang:", [[d] for d in DIRECTIONS])

            if kind == "register_direction":
                direction = text.strip()

                    return

            if kind == "task_create":

                if not title:
                    self.send(uid, "Topshiriq sarlavhasini kiriting.")
                    return

                self.write(
                    "INSERT INTO tasks(creator, title, due, region, district, direction, status) VALUES(?,?,?,?,?,?,?)",

                    self.clear_session(uid)

                body = text.strip() if text else ""

        if text == "📤 Hisobot yuborish":
            self.send(uid, "Topshiriq raqamini kiriting, masalan: /report 12")

        if command == "/report":

                return

            return
        state = session[1]

                    score,
                    len(QUESTIONS),
                    percent,
                    score,
                    percent,
                    1,
                ),
            )

def create_web_app(state):
    async def ping(request):
        return web.json_response({"service": "Koordinatorlar platformasi", "status": "ok"})

    async def health(request):
        healthy = state.get("ready", False) and time.monotonic() - state.get("last_poll", 0) < 180
        return web.json_response({"status": "ok" if healthy else "starting_or_unavailable"}, status=200 if healthy else 503)

    app = web.Application()
    app.router.add_get("/", ping)
    app.router.add_get("/health", health)
    return app


def run_polling(token, owner, db_path, directions, stop, state):
    app = App(db_path, Telegram(token), owner, directions)
    try:
        me = app.api.call("getMe")
        hook = app.api.call("getWebhookInfo")
        if hook.get("url"):
            raise RuntimeError("Webhook faollashtirilgan. Polling uchun avval webhookni o‘chiring.")
        state["ready"] = True
        print("Bot ishga tushdi: @" + me["username"], flush=True)
        offset = int((app.one("SELECT value FROM meta WHERE key='offset'") or {"value": "0"})["value"])
        last = 0
        while not stop.is_set():
            try:
                updates = app.api.call("getUpdates", offset=offset, timeout=25, allowed_updates=["message"])
                state["last_poll"] = time.monotonic()
                for update in updates:
                    if stop.is_set():
                        break
                    if "message" in update:
                        msg = update["message"]
                        if msg.get("text") and msg["text"].strip() and msg["text"].strip().upper() in {"A", "B", "C", "D"}:
                            app.handle_quiz_answer(msg["from"]["id"], msg["text"].strip())
                        else:
                            app.handle(msg)
                    offset = update["update_id"] + 1
                    app.write("INSERT OR REPLACE INTO meta(key, value) VALUES(?, ?)", ("offset", str(offset)))
                if not stop.is_set() and time.monotonic() - last > 60:
                    last = time.monotonic()
            except RuntimeError as exc:
                print(str(exc), flush=True)
                stop.wait(5)
    finally:
        state["ready"] = False
        app.db.close()


async def serve(token, owner, port, db_path, directions):
    stop = threading.Event()
    shutdown = asyncio.Event()
    state = {"ready": False, "last_poll": 0.0}
    loop = asyncio.get_running_loop()
    for sig in (signal.SIGTERM, signal.SIGINT):
        try:
            loop.add_signal_handler(sig, shutdown.set)
        except (NotImplementedError, RuntimeError):
            pass
    runner = web.AppRunner(create_web_app(state), access_log=None)
    await runner.setup()
    worker = waiter = None
    try:
        await web.TCPSite(runner, host="0.0.0.0", port=port).start()
        print(f"HTTP server: 0.0.0.0:{port}", flush=True)
        worker = asyncio.create_task(asyncio.to_thread(run_polling, token, owner, db_path, directions, stop, state))
        waiter = asyncio.create_task(shutdown.wait())
        done, _ = await asyncio.wait({worker, waiter}, return_when=asyncio.FIRST_COMPLETED)
        if worker in done:
            await worker
    finally:
        stop.set()
        if waiter:
            waiter.cancel()
        await runner.cleanup()
        if worker:
            await asyncio.shield(worker)
